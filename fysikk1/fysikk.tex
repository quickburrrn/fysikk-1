\subsubsection{Kraft $\vec{F}$}

\begin{itemize}
    \item Symbol: $\vec{F}$
    \item Oppgis i newton $N$
    \item Kan endre farten til en ting som har en masse
    \item $1N$ tilsvarer å akselerere noe på $1kg$ med $1m/s^2$
    \item $kg \cdot m/s^2 = N$ : Altså newton er masse gange akselerasjon
\end{itemize}
